函数

strlen(a) 取字符串长度，要求字符串最后要有\0

strcpy(a, b) 复制字符串,将b复制到a

strncpy(a, b, n) 复制指定长度字符串,将b的前n个字符复制至a，注意a最后手动加\0

strcat(a, b) 字符串连接, 将b跟到a后面

strncat(a, b, n) 连接指定长度字符串，将b前n个字符跟到a后面，最后手动加\0

strcmp(a, b) 比较字符串 a==b:返回0, a!=b:返回1或-1（什么时候1，什么时候-1？？？）

strcasecmp(a,b) 忽略大小写比较字符串,a==b:返回0, a!=b:1或-1

strncmp(a, b, n) 比较指定长度字符串,比较a与b的前n个字符

strchr(a,c) 在字符串中查找指定字符c，若找到则返回指定字符第一次出现的位置到字符串结束的子串

strrchr(a,c) 在字符串中反向查找定字符c,若找到则返回指定字符第一次出现的位置到字符串结束的子串

long long ksm(long long a,long long b,long long mod)
{
	if(b==0)return 1;
	long long r=ksm(a,b/2,mod);
	if(b%2)return r*r%mod*a%mod;

#include <iostream>
#include <windows.h>
#include <thread>
#include <chrono>
// 全局变量控制程序状态
bool is_running = false;    // 是否正在连点
bool is_exit = false;       // 是否退出程序
int click_interval = 50;    // 点击间隔（毫秒），默认50ms
// 模拟鼠标左键点击
void mouse_click() {
    // 按下鼠标左键
    mouse_event(MOUSEEVENTF_LEFTDOWN, 0, 0, 0, 0);
    // 释放鼠标左键
    mouse_event(MOUSEEVENTF_LEFTUP, 0, 0, 0, 0);
}
// 监听键盘按键的线程函数
void key_listener() {
    while (!is_exit) {
        // 检测 F6 键（启动/暂停）
        if (GetAsyncKeyState(VK_F6) & 0x8000) {
            is_running = !is_running;
            if (is_running) {
                std::cout << "\n[+] 连点已启动 | 点击间隔: " << click_interval << "ms | F6暂停 | F7退出" << std::endl;
            } else {
                std::cout << "\n[!] 连点已暂停 | F6继续 | F7退出" << std::endl;
            }
            // 短暂延时避免重复触发
            std::this_thread::sleep_for(std::chrono::milliseconds(300));
        }
        // 检测 F7 键（退出程序）
        if (GetAsyncKeyState(VK_F7) & 0x8000) {
            is_exit = true;
            is_running = false;
            std::cout << "\n[*] 程序即将退出..." << std::endl;
            break;
        }
        // 降低CPU占用
        std::this_thread::sleep_for(std::chrono::milliseconds(50));
    }
}
int main() {
    // 控制台编码设置（解决中文乱码）
    SetConsoleOutputCP(CP_UTF8);
    std::cout << "==================== 鼠标连点器 ====================" << std::endl;
    std::cout << "使用说明：" << std::endl;
    std::cout << "1. F6 键：启动/暂停连点" << std::endl;
    std::cout << "2. F7 键：退出程序" << std::endl;
    std::cout << "3. 可修改代码中的 click_interval 调整点击间隔（默认50ms）" << std::endl;
    std::cout << "====================================================\n" << std::endl;
    std::cout << "等待操作...（F6启动 | F7退出）" << std::endl;
    // 创建键盘监听线程
    std::thread listener_thread(key_listener);
    // 主循环：处理连点逻辑
    while (!is_exit) {
        if (is_running) {
            mouse_click();
            // 等待指定的点击间隔
            std::this_thread::sleep_for(std::chrono::milliseconds(click_interval));
        } else {
            // 未运行时降低CPU占用
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
    }
    // 等待监听线程结束
    if (listener_thread.joinable()) {
        listener_thread.join();
    }
    std::cout << "\n程序已退出！" << std::endl;
    system("pause");
    return 0;
}
